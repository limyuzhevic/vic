# NLM - Easy Usage Guide for Beginners

## What is NLM?

NLM is a **brain simulator** for computers. It lets you create virtual brains that can learn, remember things, and make decisions - just like how your brain works!

Think of it like this:
- **Neurons** = Brain cells that send signals to each other
- **Synapses** = Connections between brain cells
- **Learning** = When connections get stronger or weaker based on what happens
- **Memory** = The brain remembering patterns

That's it! NLM simulates all of this.

---

## The 3 Things You Need to Know

1. **Brain** - The virtual brain that thinks
2. **World** - The environment the brain lives in
3. **Agent** - The bridge connecting brain to world

## Quick Start (Copy & Paste)

### Example 1: Simplest Brain

```python
import pynlm

# Step 1: Make a brain
brain = pynlm.createBrain(pynlm.createDefaultConfig())

# Step 2: Turn it on
brain.initialize()

# Step 3: Make it think for 100 steps
for i in range(100):
    brain.step(i)  # One step of thinking

print("Done! Your brain thought", brain.getTotalSpikeCount(), "times")
```

### Example 2: Brain in a Simple World

```python
import pynlm

# Create everything
brain = pynlm.createBrain(pynlm.createDefaultConfig())
brain.initialize()

world = pynlm.createSimpleWorld()
world.configure(width=10, height=10, visionWidth=8, visionHeight=8)
world.reset()

agent = pynlm.createAgentBrain(brain)
agent.initialize(world)

# Run for 50 steps
for step in range(50):
    world.update(0.1)  # Update world
    percept = world.getSensoryPercept()  # What does the agent see?
    agent.processSensoryInput(percept)  # Brain sees it
    brain.step(step)  # Brain thinks
    action = agent.decodeMotorCommand()  # Brain decides action
    world.applyMotorCommand(action, world.getSimulationTime())  # Do action

print("Simulation finished!")
print("Firing neurons:", brain.getFiringNeuronCount())
```

### Example 3: Complete Agent with Learning

```python
import pynlm

def run_advanced_agent_simulation(num_steps=1000):
    """Run a complete agent simulation with learning."""
    
    # 1. Create configuration
    config = pynlm.createDefaultConfig()
    
    # 2. Create brain and initialize
    brain = pynlm.createBrain(config)
    brain.initialize()
    
    # 3. Create agent brain interface
    agent = pynlm.createAgentBrain(brain)
    
    # 4. Create and configure world
    world = pynlm.createSimpleWorld()
    world.configure(width=15, height=15, visionWidth=8, visionHeight=8)
    world.reset()
    agent.initialize(world)
    
    # 5. Enable advanced learning features
    agent.enableRewardModulation(True)      # Learn from rewards
    agent.enableStructuralPlasticity(True)  # Grow new connections
    agent.enableDevelopment(True)           # Brain matures over time
    agent.enableCuriosity(True)             # Explore new things
    
    # 6. Run simulation loop
    for step in range(num_steps):
        # Update world
        world.update(0.1)
        
        # Get sensory input from world
        percept = world.getSensoryPercept()
        
        # Process sensory input in brain
        agent.processSensoryInput(percept)
        
        # Run brain step
        brain.step(step)
        
        # Decode motor command from brain activity
        action = agent.decodeMotorCommand()
        
        # Apply action to world
        world.applyMotorCommand(action, world.getSimulationTime())
        
        # Apply reward modulation based on outcome
        reward = 0.0
        if action.getType() == pynlm.ActionType.Interact:
            # Get nearby objects and calculate reward
            percept = world.getSensoryPercept()
            if percept and percept.getInternal():
                reward = percept.getInternal()[0] if percept.getInternal() else 0.0
        
        agent.applyRewardModulation(reward, 0.0)
        
        # Update development
        agent.updateDevelopment(0.1)
        
        # Print progress with rich statistics
        if step % 100 == 0:
            print(f"Step {step}:")
            print(f"  Firing neurons: {brain.getFiringNeuronCount()}")
            print(f"  Average firing rate: {brain.getAverageFiringRate():.3f}")
            print(f"  Curiosity level: {agent.getCuriosityLevel():.3f}")
            print(f"  Novelty level: {agent.getNoveltyLevel():.3f}")
            print(f"  Development stage: {agent.getDevelopmentalStage()}")
            print(f"  Total spikes: {brain.getTotalSpikeCount()}")
    
    # Print final summary
    print("\n=== Agent Simulation Complete ===")
    print(f"Total simulation steps: {num_steps}")
    print(f"Final firing neurons: {brain.getFiringNeuronCount()}")
    print(f"Final average firing rate: {brain.getAverageFiringRate():.3f}")
    print(f"Development stage reached: {agent.getDevelopmentalStage()}")
    
    return brain, agent, world

# Run the simulation
brain, agent, world = run_advanced_agent_simulation(500)
```

### Example 4: Memory and Prediction System

```python
import pynlm
import time

def run_advanced_experiments():
    """Run advanced experiments with memory and prediction."""
    
    # Create brain with more neurons for complex experiments
    config = pynlm.createDefaultConfig()
    config.set("brain.neuron_count", 2000)
    config.set("brain.synapse_density", 0.15)
    config.set("plasticity.stdp.enable", True)
    config.set("plasticity.hebbian.enable", True)
    
    brain = pynlm.createBrain(config)
    brain.initialize()
    
    # Create agent with all systems enabled
    agent = pynlm.createAgentBrain(brain)
    world = pynlm.createSimpleWorld()
    world.configure(width=20, height=20, visionWidth=8, visionHeight=8)
    world.reset()
    agent.initialize(world)
    
    # Enable all advanced features
    agent.enableRewardModulation(True)
    agent.enableStructuralPlasticity(True)
    agent.enableDevelopment(True)
    agent.enableCuriosity(True)
    
    print("=== Advanced Experiment: Memory and Prediction ===")
    print("Running experiment with 2000 neurons...")
    print("This demonstrates:")
    print("  - Working memory (persistent activity)")
    print("  - Episodic memory (experience storage)")
    print("  - Novelty detection")
    print("  - Curiosity-driven exploration")
    print("  - Reward prediction error")
    print()
    
    # Track memory statistics
    memory_episodes = 0
    curiosity_events = 0
    prediction_errors = 0
    
    for step in range(1000):
        # Update world
        world.update(0.1)
        
        # Get sensory input
        percept = world.getSensoryPercept()
        agent.processSensoryInput(percept)
        
        # Run brain
        brain.step(step)
        
        # Decode action
        action = agent.decodeMotorCommand()
        world.applyMotorCommand(action, world.getSimulationTime())
        
        # Apply reward
        reward = world.getSensoryPercept().getInternal()[0] if world.getSensoryPercept().getInternal() else 0.0
        agent.applyRewardModulation(reward, 0.0)
        
        # Update development
        agent.updateDevelopment(0.1)
        
        # Track statistics
        if brain.getEpisodicMemory() and brain.getEpisodicMemory().getEpisodeCount() > 0:
            memory_episodes = brain.getEpisodicMemory().getEpisodeCount()
        
        if agent.getCuriosityLevel() > 0.5:
            curiosity_events++
        
        prediction_errors += agent.getPredictionError()
        
        # Print periodic updates
        if step % 200 == 0:
            print(f"Step {step}:")
            print(f"  Episodes stored: {memory_episodes}")
            print(f"  Curiosity events: {curiosity_events}")
            print(f"  Total prediction errors: {prediction_errors}")
            print(f"  Working memory traces: {brain.getWorkingMemory() and brain.getWorkingMemory().getActiveTraces() or 0}")
    
    print("\n=== Experiment Results ===")
    print(f"Total steps: 1000")
    print(f"Final episodes stored: {memory_episodes}")
    print(f"Total curiosity-driven decisions: {curiosity_events}")
    print(f"Average prediction error: {prediction_errors/1000:.3f}")
    
    return brain, agent, world

# Run the experiment
brain, agent, world = run_advanced_experiments()
```

## Troubleshooting

**"My brain isn't doing anything"**
- Did you call `brain.initialize()`?
- Try increasing the number of steps

**"The agent isn't moving"**
- Did you call `world.applyMotorCommand()`?
- Check that `world.update()` is being called

**"Everything is 0"**
- Brains need time to "warm up" - try more steps
- Some neurons need input to fire - make sure sensory input is connected
