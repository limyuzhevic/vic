# NLM Basic Example: Running a Simple Simulation

This example demonstrates how to create and run a basic NLM simulation with advanced configuration management.

## Example 1: Basic Simulation with Configuration File

```cpp
#include "core/Config/Config.hpp"
#include "core/Logger/Logger.hpp"
#include "brain/Brain.hpp"
#include <iostream>

int main() {
    // Configure logging
    nlm::Logger::configureGlobalLogger(nlm::LogLevel::Info, true);
    
    // Load configuration
    auto config = std::make_shared<nlm::Config>();
    if (!config->loadFromFile("config.yaml")) {
        std::cerr << "Failed to load configuration! Using defaults." << std::endl;
        // Set default values
        config->set("neuron_count", 1000LL, nlm::ConfigSource::Default);
        config->set("simulation_timestep", 0.001, nlm::ConfigSource::Default);
        config->set("connection_probability", 0.15f, nlm::ConfigSource::Default);
    }
    
    // Validate configuration
    if (!config->validate()) {
        std::cerr << "Configuration validation failed!" << std::endl;
        return 1;
    }
    
    // Create brain
    auto brain = std::make_shared<nlm::Brain>(config);
    if (!brain->initialize()) {
        std::cerr << "Failed to initialize brain!" << std::endl;
        return 1;
    }
    
    // Run simulation
    brain->logStatus();
    
    // Run basic tests
    runBasicConnectivityTest(brain);
    
    // Run plasticity experiment
    brain->reset();
    brain->initialize();
    runPlasticityExperiment(brain);
    
    std::cout << "Simulation completed successfully!" << std::endl;
    return 0;
}
```

## Example 2: Agent-Based Simulation

```cpp
#include "agent/AgentBrain.hpp"
#include "agent/SensoryPercept.hpp"
#include "world/SimpleWorld.hpp"
#include <iostream>

int main() {
    // Initialize logger
    nlm::Logger::configureGlobalLogger(nlm::LogLevel::Info, true);
    
    // Create brain configuration
    auto config = std::make_shared<nlm::Config>();
    config->set("neuron_count", 500LL, nlm::ConfigSource::Default);
    config->set("simulation_timestep", 0.001, nlm::ConfigSource::Default);
    
    // Create brain
    auto brain = std::make_shared<nlm::Brain>(config);
    
    // Create agent brain
    auto agentBrain = std::make_shared<nlm::AgentBrain>(brain);
    
    // Initialize world
    nlm::SimpleWorld world(32, 32);
    agentBrain->initialize(world);
    
    // Create sensory percept
    nlm::SensoryPercept percept;
    // ... populate percept with sensor data ...
    
    // Process sensory input
    agentBrain->processSensoryInput(percept);
    
    // Get motor command
    auto command = agentBrain->decodeMotorCommand();
    std::cout << "Motor command: " << static_cast<int>(command) << std::endl;
    
    // Update development
    agentBrain->updateDevelopment(0.001);
    
    // Apply reward modulation
    agentBrain->applyRewardModulation(1.0f, 0.5f);
    
    std::cout << "Agent simulation completed!" << std::endl;
    return 0;
}
```

## Example 5: Error Handling and Validation

```cpp
#include "core/Config/Config.hpp"
#include "core/Logger/Logger.hpp"
#include <iostream>

int main() {
    // Configure logging
    nlm::Logger::configureGlobalLogger(nlm::LogLevel::Info, true);
    
    try {
        // Load configuration
        auto config = std::make_shared<nlm::Config>();
        if (!config->loadFromFile("config.yaml")) {
            throw std::runtime_error("Failed to load configuration file");
        }
        
        // Validate configuration
        if (!config->validate()) {
            throw std::runtime_error("Configuration validation failed");
        }
        
        // Set configuration values
        config->set("neuron_count", 1000LL, nlm::ConfigSource::Runtime);
        config->set("simulation_timestep", 0.001, nlm::ConfigSource::Runtime);
        
        // Save configuration
        if (!config->saveToFile("validated_config.yaml")) {
            throw std::runtime_error("Failed to save configuration");
        }
        
        std::cout << "Configuration processed successfully!" << std::endl;
        
    } catch (const std::exception& e) {
        std::cerr << "Error: " << e.what() << std::endl;
        return 1;
    }
    
    return 0;
}
```

## Example 6: Log Management

```cpp
#include "core/Logger/Logger.hpp"
#include <iostream>

int main() {
    // Configure logging with rotation
    nlm::Logger::configureGlobalLogger(nlm::LogLevel::Info, true);
    nlm::Logger::addGlobalLogFile("nlm.log", true);
    
    // Log different types of messages
    NLM_LOG_INFO("Application started");
    NLM_LOG_DEBUG("Debug information for troubleshooting");
    NLM_LOG_WARNING("Warning: Low memory");
    NLM_LOG_ERROR("Error: File not found");
    NLM_LOG_CRITICAL("Critical: System failure");
    
    // Log structured data
    NLM_LOG_INFO("Experiment with " << 1000 << " neurons");
    NLM_LOG_INFO("Success rate: " << 0.95f * 100 << "%");
    
    // Configure log rotation
    auto& logger = nlm::Logger::getGlobal();
    for (auto& log : logger.pImpl->loggers) {
        auto consoleLogger = std::dynamic_pointer_cast<nlm::ConsoleLogger>(log);
        if (consoleLogger) {
            consoleLogger->enableLogRotation(true);
            consoleLogger->setMaxFileSize(50 * 1024 * 1024); // 50MB
            break;
        }
    }
    
    std::cout << "Logging configured and operational!" << std::endl;
    return 0;
}
```

## Troubleshooting Guide

### Symptom: Configuration file not found
**Solution:** Use default configuration or specify correct file path.
```cpp
if (!config->loadFromFile("config.yaml")) {
    NLM_LOG_INFO("Using default configuration.");
    // Set default values
}
```

### Symptom: Log files growing indefinitely
**Solution:** Enable log rotation.
```cpp
nlm::Logger::addGlobalLogFile("nlm.log", true);
```

### Symptom: Memory usage high
**Solution:** Use smart pointers and validate memory usage.
```cpp
auto brain = std::make_shared<Brain>(config);
```

### Symptom: Configuration validation fails
**Solution:** Validate configuration before use.
```cpp
if (!config->validate()) {
    NLM_LOG_ERROR("Configuration validation failed");
    // Fix configuration or exit
}
```

### Symptom: No output or silent failure
**Solution:** Configure logging and check for errors.
```cpp
nlm::Logger::configureGlobalLogger(nlm::LogLevel::Info, true);
```
