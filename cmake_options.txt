# NLM Build Configuration Example
# This file demonstrates common CMake configuration patterns
# Save as cmake_options.txt or include in documentation

# =============================================================================
# Basic Build Configuration
# =============================================================================

# Basic project setup
cmake_minimum_required(VERSION 3.16)
project(NLM VERSION 0.1.0 LANGUAGES CXX)

# C++20 standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Default build type
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# =============================================================================
# Build Options
# =============================================================================

option(ENABLE_OPENMP "Enable OpenMP parallelization" ON)
option(ENABLE_AVX2 "Enable AVX2 instructions" ON)
option(ENABLE_SSE4_2 "Enable SSE4.2 instructions" ON)
option(ENABLE_MEMORY_POOLS "Enable memory pools for performance" ON)
option(BUILD_SHARED_LIBS "Build shared libraries instead of static" OFF)
option(BUILD_PYTHON_BINDINGS "Build Python bindings" ON)
option(BUILD_TESTS "Build test suite" ON)
option(INSTALL_DEMOS "Install demo executables" ON)

# =============================================================================
# Compiler-Specific Settings
# =============================================================================

# Warning flags
if(MSVC)
    add_compile_options(/W4)
else()
    add_compile_options(-Wall -Wextra -Wpedantic -Wshadow -Wnon-virtual-dtor)
endif()

# SIMD instructions
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    if(ENABLE_AVX2)
        add_compile_options(-mavx2)
    endif()
    if(ENABLE_SSE4_2)
        add_compile_options(-msse4.2)
    endif()
elseif(MSVC)
    if(ENABLE_AVX2)
        add_compile_options(/arch:AVX2)
    endif()
endif()

# OpenMP
if(ENABLE_OPENMP)
    find_package(OpenMP REQUIRED)
    if(OpenMP_CXX_FOUND)
        add_compile_options(${OpenMP_CXX_FLAGS})
        link_libraries(${OpenMP_CXX_FLAGS})
    endif()
endif()

# =============================================================================
# Performance Optimizations
# =============================================================================

# Memory pools
if(ENABLE_MEMORY_POOLS)
    add_compile_definitions(ENABLE_MEMORY_POOLS)
    add_compile_options(-DENABLE_MEMORY_POOLS)
endif()

# Optimization levels by build type
set(CMAKE_CXX_FLAGS_DEBUG "-g -O0 -DDEBUG -DNDEBUG")
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
set(CMAKE_CXX_FLAGS_RELWITHDEBINFO "-O2 -g -DNDEBUG")
set(CMAKE_CXX_FLAGS_MINSIZEREL "-Os -DNDEBUG")

# =============================================================================
# Target Configuration
# =============================================================================

# Include directories
target_include_directories(nlm_core PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
    $<INSTALL_INTERFACE:include>
)

# Install targets
install(TARGETS nlm nlm_core nlm_agent nlm_world nlm_phase3 nlm_phase4
    ARCHIVE DESTINATION lib
    LIBRARY DESTINATION lib
    RUNTIME DESTINATION bin
)

install(DIRECTORY src/
    DESTINATION include/nlm
    FILES_MATCHING PATTERN "*.hpp"
    PATTERN "*.h"
)

# =============================================================================
# Python Bindings Configuration
# =============================================================================

if(BUILD_PYTHON_BINDINGS)
    enable_language(PYTHON)
    find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
    
    # Python package configuration
    set(PYTHON_MODULE_NAME pynlm)
    set(PYTHON_MODULE_SO_NAME pynlm)
    
    # Additional Python-specific settings
    add_compile_definitions(PYBIND11_DETAILED_ERROR_MESSAGES)
endif()

# =============================================================================
# Installation Examples
# =============================================================================

# System installation (requires root)
# cmake .. -DCMAKE_BUILD_TYPE=Release -DCMAKE_INSTALL_PREFIX=/usr/local
# make install

# User installation (no root required)
# cmake .. -DCMAKE_BUILD_TYPE=Release -DCMAKE_INSTALL_PREFIX=~/.local
# make install

# Development installation (with symbols)
# cmake .. -DCMAKE_BUILD_TYPE=Debug -DCMAKE_INSTALL_PREFIX=/usr/local
# make install

# Usage Examples
# =============================================================================

# Example: Debug build with tests

# Example: Minimal build for deployment
# cmake .. -DCMAKE_BUILD_TYPE=MinSizeRel -DBUILD_TESTS=OFF -DBUILD_PYTHON_BINDINGS=OFF

# =============================================================================
# Performance Monitoring
# =============================================================================

# Add performance monitoring if available
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    add_compile_definitions(ENABLE_PERFORMANCE_MONITORING)
endif()

# End of Configuration
