// CMakeLists.txt updates for JSON library support

# Find nlohmann_json library (header-only)
find_package(nlohmann_json QUIET)
if (NOT nlohmann_json_FOUND)
    # Try to find it as a package
    find_package(nlohmann_json REQUIRED)
else()
    # If not found, include the header directly (assuming it's available in system)
    message(STATUS "Using system nlohmann_json library")
endif()
